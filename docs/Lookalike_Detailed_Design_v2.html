1.3 Design Principles

  1. Score-Threshold Separation: Threshold is NOT a process creation input; it's a post-scoring dashboard analysis tool (BRD §6.1)
  2. Full Population Scoring: Score 100% of valid candidates; never silently discard low-score customers (BRD §6.4.4)
  3. Version Immutability: Each run creates an independent version with frozen config snapshot
  4. Temporal Leakage Prevention: Mandatory audit step excludes future-information fields before training (BRD §3.4)
  5. Dual-Model Independence: If multi-model scenario is enabled, each model executes independently (BRD §6.4.2)
  6. Cold Start Exclusion: Customers with no transaction history are not covered by Lookalike (BRD §1.3)
  7. No Fixed Metric Thresholds: AUC/KS are diagnostic, not pass/fail gates; business validation via A/B test is ultimate success measure (BRD §5.2)

5.1 Same-CIF Transaction Exclusion

def exclude_same_cif_transactions(txn_records, customer_cif):
    """Exclude intra-bank transfers between accounts under the same CIF."""
    account_ids = get_accounts_by_cif(customer_cif)
    filtered = txn_records.filter(
        NOT (sender_account IN account_ids AND receiver_account IN account_ids)
    )
    return filtered
    # Note: Third-party and cross-bank transactions are NOT excluded

5.4 Temporal Leakage Audit (Mandatory)

CRITICAL: Temporal leakage occurs when the model uses future information during training. This audit MUST be performed before every training run.
Field to ExcludeReason
Current loan balanceOnly exists after loan disbursement — post-event information
Recent disbursement dateOnly exists after disbursement — post-event
Marketing touch outcomeOnly known after customer is reached
Credit decision scoreOutput of another model — should not be an input
Customer ID, ID numberBusiness identifiers — no predictive value
Eligibility criteria fieldsAll candidates have same value (all eligible) — zero discriminatory power within group
Key Distinction: Eligibility criteria (age, nationality) determine who enters the candidate pool. Model features determine who is more quality within the pool. Must not be mixed.

5.5 WOE Encoding & IV Calculation

def compute_woe_iv(df, feature, target, bins=10):
    """Compute Weight of Evidence and Information Value."""
    grouped = df.groupby('bin')[target].agg(['count', 'sum'])
    grouped['non_event'] = grouped['count'] - grouped['sum']
    grouped['positive_recall'] = grouped['sum'] / grouped['sum'].sum()
    grouped['negative_recall'] = grouped['non_event'] / grouped['non_event'].sum()
    grouped['woe'] = np.log(grouped['positive_recall'] / grouped['negative_recall'])
    grouped['iv'] = (grouped['positive_recall'] - grouped['negative_recall']) * grouped['woe']
    iv = grouped['iv'].sum()
    # KS Table fields: positive_recall, negative_recall, cum_positive_recall, cum_negative_recall, ks
    return woe_map, iv

# High (4-5 stars): Prioritise for extraction, subject to actual IV

# Low (1-2 stars): Monitor; exclude if IV < 0.02

ElementDefinition

6.6 Training Pipeline (Pseudo-code)

class LookalikeModelTrainer:
    def train_product_model(self, product: str):
        # 1. Load seed + negative samples (product-specific)
        seed = self.load_seed_samples(product)
        negative = self.load_negative_samples(product)
        
        # 2. Feature engineering (71 core + optional 24 for mortgage)
        features = self.feature_engineering(
            seed + negative, feature_list=FEATURE_CATALOG[product],
            exclude_same_cif=True, apply_quality_screening=True,
            temporal_leakage_audit=True
        )
        
        # 3. Dataset split (60/20/20 stratified)
        train, val, test = self.stratified_split(features, ratios=[0.6, 0.2, 0.2])
        oot = self.load_oot_samples(product, time_window=self.oot_window)
        
        # 4. WOE encoding (for LR)
        woe_maps = self.compute_woe(train)
        
        # 5. Train LR + WOE (baseline)
        lr_model = LogisticRegression()
        lr_model.fit(self.apply_woe(train, woe_maps))
        lr_metrics = self.evaluate(lr_model, val, test, oot)
        
        # 6. Train LightGBM (primary)
        lgb_model = LGBMClassifier(
            n_estimators=500, learning_rate=0.05, max_depth=6,
            num_leaves=31, subsample=0.8, colsample_bytree=0.8,
            class_weight='balanced'
        )
        lgb_model.fit(train, eval_set=[(val,)], early_stopping_rounds=50)
        lgb_metrics = self.evaluate(lgb_model, val, test, oot)
        
        # 7. Save to model registry
        self.save_to_registry(product, lr_model, lgb_model, woe_maps, 
                              metrics, feature_list, training_config)
        return {"lr": lr_metrics, "lgb": lgb_metrics}

class FeatureEngineeringService:
    def apply_quality_screening(self, df):
        """Apply BRD-defined data quality rules."""
        for col in df.columns:
            missing_rate = df[col].isna().mean()
            single_value_ratio = df[col].value_counts(
                normalize=True).iloc[0] if not df[col].isna().all() else 1
            
            if missing_rate > 0.95:    # Rule 1: >95% missing -> exclude
                df = df.drop(columns=[col])
                continue
            if single_value_ratio > 0.95:  # Rule 2: >95% single value -> exclude
                df = df.drop(columns=[col])
                continue
            # Rule 3: Impute missing values
            if missing_rate > 0:
                if df[col].dtype in ['float64', 'int64']:
                    df[col] = df[col].fillna(df[col].median())
                else:
                    df[col] = df[col].fillna(df[col].mode()[0])
        return df

// Histogram pre-computed during scoring, cached in Redis

const ScoreDistributionChart = ({ versionId }) => {
  const { data: histogram } = useQuery(['histogram', versionId],
    () => fetchHistogram(versionId));
  const [threshold, setThreshold] = useState(0.5);
  
  // Local calculation: sum counts where bin >= threshold
  const selectedCount = useMemo(() => {
    return histogram.counts.reduce((sum, count, i) =>
      histogram.bins[i] >= threshold ? sum + count : sum, 0);
  }, [histogram, threshold]);
  
  return (
    <Card>
      <HistogramChart data={histogram} threshold={threshold} />
      <Slider min={0} max={1} step={0.01} value={threshold}
        onChange={setThreshold}  // No API call!
        tooltip={{ formatter: v => `>= ${v.toFixed(2)}` }} />
      <Statistic title="Selected Customers" value={selectedCount} />
    </Card>
  );
};

14.1 PSI Monitoring Implementation

class PSIMonitor:
    def compute_psi(self, train_dist, current_dist, bins=10):
        """Calculate Population Stability Index."""
        train_pct = train_dist / train_dist.sum()
        current_pct = current_dist / current_dist.sum()
        train_pct = np.where(train_pct == 0, 0.0001, train_pct)
        current_pct = np.where(current_pct == 0, 0.0001, current_pct)
        psi = np.sum((current_pct - train_pct) * np.log(current_pct / train_pct))
        return psi

    def monitor_all_features(self, model_id):
        """Monthly PSI check for all features of a deployed model."""
        model = self.registry.get_model(model_id)
        results = []
        for feature in model.feature_list:
            psi = self.compute_psi(
                model.train_feature_distributions[feature],
                self.extract_current_distributions(feature, period=current_month())
            )
            results.append({
                'feature': feature, 'psi': psi,
                'status': 'stable' if psi <= 0.10 else 'monitor' if psi <= 0.25 else 'unstable',
                'triggers_retrain': psi > 0.25
            })
        unstable = [r for r in results if r['triggers_retrain']]
        if unstable:
            self.alert_ds_team(model_id, unstable)
            retrain_task.delay(model_id)  # Auto-trigger retrain
